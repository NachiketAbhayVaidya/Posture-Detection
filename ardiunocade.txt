// =====================================================
// TIME FORMAT
// =====================================================

String formatTime(long sec) {

  int m = sec / 60;
  int s = sec % 60;

  char buf[10];

  sprintf(buf, "%02d:%02d", m, s);

  return String(buf);
}

// =====================================================
// VIBRATION
// =====================================================

void vibe1Start(int durationMs) {

  digitalWrite(VIBE_PIN_1, HIGH);

  vibe1UntilMs = millis() + durationMs;
}

void vibe2Start(int durationMs) {

  digitalWrite(VIBE_PIN_2, HIGH);

  vibe2UntilMs = millis() + durationMs;
}

void vibeBothStart(int durationMs) {

  vibe1Start(durationMs);
  vibe2Start(durationMs);
}

void vibeUpdate() {

  if (millis() >= vibe1UntilMs) {
    digitalWrite(VIBE_PIN_1, LOW);
  }

  if (millis() >= vibe2UntilMs) {
    digitalWrite(VIBE_PIN_2, LOW);
  }
}

void vibePulses(int count, int durMs, int gapMs) {

  for (int i = 0; i < count; i++) {

    digitalWrite(VIBE_PIN_1, HIGH);
    digitalWrite(VIBE_PIN_2, HIGH);

    delay(durMs);

    digitalWrite(VIBE_PIN_1, LOW);
    digitalWrite(VIBE_PIN_2, LOW);

    if (i < count - 1) {
      delay(gapMs);
    }
  }
}

// =====================================================
// SEVERITY
// =====================================================

String getSeverity(float p) {

  float absP = abs(p);

  if (absP < SLOUCH_THRESHOLD) {
    return "Good";
  }
  else if (absP < SLOUCH_THRESHOLD + 8) {
    return "Mild";
  }
  else if (absP < SLOUCH_THRESHOLD + 18) {
    return "Moderate";
  }
  else {
    return "Severe";
  }
}

int getSeverityScore(float p) {

  float absP = abs(p);

  if (absP <= 0) {
    return 10;
  }

  if (absP >= SLOUCH_THRESHOLD + 25) {
    return 0;
  }

  float score =
      10.0 - (absP / (SLOUCH_THRESHOLD + 25.0)) * 10.0;

  return constrain((int)score, 0, 10);
}

// =====================================================
// READ FSR
// =====================================================

void readFSR() {

  fsrLeft  = analogRead(FSR_LEFT);
  fsrRight = analogRead(FSR_RIGHT);

  isSitting =
      (fsrLeft > FSR_THRESHOLD) &&
      (fsrRight > FSR_THRESHOLD);
}

// =====================================================
// READ PITCH
// =====================================================

float readPitch() {

  int16_t ax, ay, az;

  mpu.getAcceleration(&ax, &ay, &az);

  float AccX = ax / 16384.0;
  float AccY = ay / 16384.0;
  float AccZ = az / 16384.0;

  float p = atan2(
                -AccX,
                sqrt(AccY * AccY + AccZ * AccZ)
            ) * 57.2958;

  if (isnan(p)) {
    p = 0;
  }

  return p;
}

// =====================================================
// PREDICTIVE WARNING
// =====================================================

void checkPredictiveWarning() {

  if (slouchConfirmed) {

    worseningCount = 0;

    previousPitch = pitch;

    return;
  }

  float rateOfChange =
      abs(pitch) - abs(previousPitch);

  if (rateOfChange > PREDICT_THRESHOLD) {
    worseningCount++;
  }
  else {
    worseningCount = 0;
  }

  if (
      worseningCount >= PREDICT_FRAMES &&
      abs(pitch) < SLOUCH_THRESHOLD
  ) {

    vibeBothStart(120);

    tone(BUZZER_PIN, 1500);

    delay(200);

    noTone(BUZZER_PIN);

    worseningCount = 0;
  }

  previousPitch = pitch;
}

// =====================================================
// FIRE ALERT
// =====================================================

void fireSlouchAlert() {

  tone(BUZZER_PIN, 2000);

  if (severityLabel == "Mild") {
    vibeBothStart(200);
  }
  else if (severityLabel == "Moderate") {
    vibeBothStart(400);
  }
  else if (severityLabel == "Severe") {
    vibeBothStart(600);
  }
}

// =====================================================
// PROCESS POSTURE
// =====================================================

void processPosture() {


  bool badPostureNow = false;

  if (pitch > SLOUCH_THRESHOLD) {

    postureLabel = "slouch";

    badPostureNow = true;
  }
  else if (pitch < -10) {

    postureLabel = "recline";

    badPostureNow = true;
  }
  else {

    postureLabel = "upright";

    badPostureNow = false;
  }

//  if (
//      abs(pitch) > 8 &&
//      abs(pitch) < SLOUCH_THRESHOLD
//  ) {
//
//    String mlGuess = runMLInference();
//
//    if (mlGuess == "slouch") {
//
//      postureLabel = "slouch";
//
//      badPostureNow = true;
//    }
//  }

  if (badPostureNow) {

    if (!slouchTimerRunning) {

      slouchTimerStartMs = millis();

      slouchTimerRunning = true;
    }

    if (!slouchDetected) {
      slouchDetected = true;

      slouchConfirmed = false;

      slouchStartMs = millis();
    }

    if (
        !slouchConfirmed &&
        (millis() - slouchStartMs >= SLOUCH_CONFIRM_MS)
    ) {

      slouchConfirmed = true;

      slouchCount++;

      fireSlouchAlert();

      if (
          pomoRunning &&
          !pausedBySlouch &&
          slouchCount >= MAX_SLOUCHES_BEFORE_PAUSE
      ) {

        pausedBySlouch = true;

        pomoRunning = false;

        vibePulses(3, 150, 100);
      }
    }

    static unsigned long lastVibeMs = 0;

    if (slouchConfirmed) {

      if (millis() - lastVibeMs >= 1500) {

        vibeBothStart(400);

        lastVibeMs = millis();
      }
    }
  }
  else {

    // STOP SLOUCH TIMER

  if (slouchTimerRunning) {

    totalSlouchTimeMs +=
        millis() - slouchTimerStartMs;

    slouchTimerRunning = false;
  }

  slouchDetected = false;

  slouchConfirmed = false;

  slouchStartMs = 0;

  noTone(BUZZER_PIN);

  severityLabel = "Good";
  }

  if (badPostureNow) {

    severityLabel = getSeverity(pitch);
  }
}

// =====================================================
// POMODORO
// =====================================================

void updatePomodoro() {

  if (!pomoRunning || pausedBySlouch) {
    return;
  }

  if (millis() - lastPomoTickMs < 1000) {
    return;
  }

  lastPomoTickMs = millis();

  if (remainingSec > 0) {

    remainingSec--;
  }
  else {

    if (!onBreak) {

       onBreak = true;

  remainingSec = BREAK_MINUTES * 60;

  // Dashboard message
  pomoStatusMessage = "Break Started";

  sessionJustChanged = true;

  // Vibrate motors
  vibePulses(2, 500, 200);
    }
    else {

      onBreak = false;

  remainingSec = FOCUS_MINUTES * 60;

  // Dashboard message
  pomoStatusMessage = "Focus Session Started";

  sessionJustChanged = true;

  // Vibrate motor
  vibePulses(1, 600, 0);
  }
    }
  }

// =====================================================
// SETUP
// =====================================================

void setup() {

  Serial.begin(115200);

  Wire.begin(21, 22);

  mpu.initialize();

  if (!mpu.testConnection()) {

    while (1);
  }

  pinMode(BUZZER_PIN, OUTPUT);

  noTone(BUZZER_PIN);

  pinMode(VIBE_PIN_1, OUTPUT);
  pinMode(VIBE_PIN_2, OUTPUT);

  digitalWrite(VIBE_PIN_1, LOW);
  digitalWrite(VIBE_PIN_2, LOW);

  vibePulses(2, 150, 100);

  lastPomoTickMs = millis();
  pomoRunning = true;
//  if (isSitting && !pomoRunning) {
//    pomoRunning = true;
//    lastPomoTickMs = millis();
//  }
}

// =====================================================
// LOOP
// =====================================================

void loop() {

  vibeUpdate();

  if (Serial.available()) {

    char cmd = Serial.read();

    if (cmd == 'r') {

      pausedBySlouch = false;

      pomoRunning = true;

      slouchCount = 0;
    }

    if (cmd == 't') {

      vibePulses(3, 200, 100);
    }
  }

  // ==========================================
  // READ FSR
  // ==========================================

  readFSR();

  // ==========================================
  // NOT SITTING
  // ==========================================

  if (!isSitting) {

    noTone(BUZZER_PIN);

    digitalWrite(VIBE_PIN_1, LOW);
    digitalWrite(VIBE_PIN_2, LOW);

    slouchDetected = false;
    slouchConfirmed = false;

    slouchStartMs = 0;

    sendDashboardData();

    delay(300);

    return;
  }

  // ==========================================
  // READ PITCH
  // ==========================================

  pitch = readPitch();

  // ==========================================
  // PROCESS
  // ==========================================

  checkPredictiveWarning();

  processPosture();

  updatePomodoro();

  // ==========================================
  // SEND DATA
  // ==========================================

  sendDashboardData();

  delay(300);
}
